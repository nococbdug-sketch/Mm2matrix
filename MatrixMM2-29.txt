local function tw(obj, time, props, style, dir)
	local t = TweenService:Create(obj, TweenInfo.new(time, style or Enum.EasingStyle.Quint, dir or Enum.EasingDirection.Out), props)
	t:Play()
	return t
end

local function new(class, props, parent)
	local o = Instance.new(class)
	for k, v in pairs(props) do
		o[k] = v
	end
	o.Parent = parent
	return o
end

local function round(o, r)
	return new("UICorner", { CornerRadius = UDim.new(0, r) }, o)
end

local function pill(o)
	return new("UICorner", { CornerRadius = UDim.new(1, 0) }, o)
end

local function edge(o, color, thickness)
	return new("UIStroke", {
		ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
		Color = color or T.stroke,
		Thickness = thickness or 1,
	}, o)
end

-- color gradient (the object must be white so the gradient colors show as-is)
local function grad(o, c0, c1, rot)
	return new("UIGradient", { Color = ColorSequence.new(c0, c1), Rotation = rot or 90 }, o)
end

-- thin inner border inside a CanvasGroup (a normal stroke would be clipped)
local function rim(parent, radius, color)
	local f = new("Frame", {
		Name = "Rim",
		Size = UDim2.new(1, -2, 1, -2),
		Position = UDim2.fromOffset(1, 1),
		BackgroundTransparency = 1,
		ZIndex = 50,
	}, parent)
	round(f, radius)
	edge(f, color or T.stroke:Lerp(WHITE, 0.1), 1)
end

local function pointer(input)
	return input.UserInputType == Enum.UserInputType.Touch
		or input.UserInputType == Enum.UserInputType.MouseButton1
end

-- text color that contrasts with color c
local function inkFor(c)
	local lum = 0.299 * c.R + 0.587 * c.G + 0.114 * c.B
	return lum > 0.55 and T.ink or WHITE
end

local function hex(c)
	return string.format(
		"#%02X%02X%02X",
		math.floor(c.R * 255 + 0.5),
		math.floor(c.G * 255 + 0.5),
		math.floor(c.B * 255 + 0.5)
	)
end

-- show a Roblox image: tries the asset id first, then the thumbnail of that asset
-- (works for decal ids too). fallback = object shown only if nothing loads.
local function setImage(img, id, fallback)
	img.Image = "rbxassetid://" .. id
	task.delay(3, function()
		if img.Parent and not img.IsLoaded then
			img.Image = "rbxthumb://type=Asset&id=" .. id .. "&w=420&h=420"
			task.delay(3, function()
				if img.Parent and not img.IsLoaded and fallback then
					fallback.Visible = true
				end
			end)
		end
	end)
end

-- text size (used for notification height and section label width)
local function textSize(text, size, font, width)
	local ok, v = pcall(function()
		return TextService:GetTextSize(text, size, font, Vector2.new(width, 10000))
	end)
	if ok and v then
		return v
	end
	local w = #text * size * 0.55
	local lines = math.max(1, math.ceil(w / math.max(width, 1)))
	return Vector2.new(math.min(w, width), lines * (size + 2))
end

-- layered glow: each layer = transparent frame + thick UIStroke outside the edge
-- layers = { {thickness, transparency}, ... }
-- limit = how many layers may show (used by lite mode)
local function addGlow(parent, radius, color, layers, on)
	local list = {}
	for _, l in ipairs(layers) do
		local f = new("Frame", {
			Name = "Glow",
			Size = UDim2.fromScale(1, 1),
			BackgroundTransparency = 1,
			BorderSizePixel = 0,
		}, parent)
		new("UICorner", { CornerRadius = radius }, f)
		local s = edge(f, color, l[1])
		s.Transparency = on and l[2] or 1
		list[#list + 1] = { stroke = s, a = l[2] }
	end
	local api = { list = list, on = on and true or false, limit = #list }
	function api:Set(v, time)
		self.on = v and true or false
		for i, g in ipairs(list) do
			local target = (self.on and i <= self.limit) and g.a or 1
			if time and time > 0 then
				tw(g.stroke, time, { Transparency = target })
			else
				g.stroke.Transparency = target
			end
		end
	end
	function api:SetLimit(n, time)
		self.limit = n
		self:Set(self.on, time)
	end
	return api
end

local function cleanup()
	local places = {}
	pcall(function() table.insert(places, gethui()) end)
	pcall(function() table.insert(places, game:GetService("CoreGui")) end)
	table.insert(places, Players.LocalPlayer:WaitForChild("PlayerGui"))
	for _, p in ipairs(places) do
		for _, n in ipairs({ GUI_NAME, GUI_NAME .. "Intro" }) do
			local old = p:FindFirstChild(n)
			if old then
				old:Destroy()
			end
		end
	end
end

local function mount(g)
	local ok = pcall(function()
		g.Parent = (gethui and gethui()) or game:GetService("CoreGui")
	end)
	if not ok or not g.Parent then
		g.Parent = Players.LocalPlayer:WaitForChild("PlayerGui")
	end
end

-- hide Roblox core UI during the intro; returns a restore function
local CORE_TYPES = {
	Enum.CoreGuiType.PlayerList,
	Enum.CoreGuiType.Health,
	Enum.CoreGuiType.Backpack,
	Enum.CoreGuiType.Chat,
	Enum.CoreGuiType.EmotesMenu,
}

local function lockCore()
	local saved, restored = {}, false
	for _, t in ipairs(CORE_TYPES) do
		local ok, v = pcall(function()
			return StarterGui:GetCoreGuiEnabled(t)
		end)
		if ok then
			saved[#saved + 1] = { t, v }
			pcall(function()
				StarterGui:SetCoreGuiEnabled(t, false)
			end)
		end
	end
	return function()
		if restored then
			return
		end
		restored = true
		for _, s in ipairs(saved) do
			pcall(function()
				StarterGui:SetCoreGuiEnabled(s[1], s[2])
			end)
		end
	end
end

local function randText(n)
	local t = {}
	for i = 1, n do
		local k = math.random(1, #CHARS)
		t[i] = string.sub(CHARS, k, k)
	end
	return table.concat(t, "\n")
end

		BorderSizePixel = 0,

	local t0 = os.clock()

	-- input blocker (stops taps from reaching the game); also used to skip
	local skip = new("TextButton", {
		Size = UDim2.fromScale(1, 1),
		BackgroundTransparency = 1,
		AutoButtonColor = false,
		Text = "",
		ZIndex = 30,
	}, root)
	skip.Activated:Connect(function()

		end
	end)

	ig.Destroying:Connect(function()
		running = false
		if hb then
			hb:Disconnect()
			hb = nil
		end
		pcall(function() spin:Cancel() end)
		pcall(function() shineTween:Cancel() end)
		restoreCore()
	end)

	finish = function()
		if finished then
			return
		end
		finished = true
		running = false
		if hb then
			hb:Disconnect()
			hb = nil
		end
		spin:Cancel()
		shineTween:Cancel()

		fill.Size = UDim2.fromScale(1, 1)
		edgeBar.Size = UDim2.new(1, 0, 0, 3)
		pct.Text = "100%"
		status.Text = "Ready"
		titleLbl.Text = FULL
		hint.Text = ""
		if #logLines > 0 then
			logLines[#logLines].done = true

		tw(seam, 0.5, { Size = UDim2.new(1, 0, 0, 2) })

		task.delay(0.5, function()
			if not ig.Parent then
				return
			end
			content.Visible = false
			overlay.Visible = false
			restoreCore()
			if o.onOpen then
				task.spawn(o.onOpen)
			end
			tw(panelTop, 0.8, { Position = UDim2.fromScale(0, -0.5) }, Enum.EasingStyle.Quint, Enum.EasingDirection.InOut)
			tw(panelBot, 0.8, { Position = UDim2.fromScale(0, 1) }, Enum.EasingStyle.Quint, Enum.EasingDirection.InOut)
			tw(seam, 0.3, { BackgroundTransparency = 1 })
			task.delay(0.85, function()
				if ig.Parent then
					ig:Destroy()
				end
				if o.onDone then
					task.spawn(o.onDone)
				end
			end)
		end)
	end

	mount(ig)

	-- "decode" title: random letters lock in one by one, with a brief glitch now and then
	local nextGlitch = t0 + 3
	local glitchUntil, glitchA, glitchB = 0, 1, 1
	local function buildTitle(now, elapsed)
		local lockedN = math.clamp(math.floor((elapsed - 0.35) / 0.1), 0, nTitle)
		local glitching = now < glitchUntil
		if lockedN >= nTitle and not glitching then
			if titleLbl.Text ~= FULL then
				titleLbl.Text = FULL
			end
			return
		end
		local out = {}
		for i = 1, nTitle do
			local ch = string.sub(FULL, i, i)
			if ch == " " then
				out[i] = ch
			elseif i > lockedN or (glitching and (i == glitchA or i == glitchB)) then
				local k = math.random(1, #GLYPHS)
				out[i] = string.sub(GLYPHS, k, k)
			else
				out[i] = ch
			end
		end
		titleLbl.Text = table.concat(out)
	end

	-- one loop for everything: progress, stages, percent, countdown, title
	local stage, lastPct, lastSec, lastTitle = 0, -1, -1, 0
	local skipText = (o.skippable ~= false) and "Tap to skip  -  " or "Loading  -  "
	hb = RunService.Heartbeat:Connect(function()
		if not running then
			return
		end
		local now = os.clock()
		local elapsed = now - t0
		local x = math.clamp(elapsed / total, 0, 1)
		local p = math.clamp(x + 0.02 * math.sin(x * math.pi * 6), 0, 1)

		fill.Size = UDim2.fromScale(p, 1)
		edgeBar.Size = UDim2.new(p, 0, 0, 3)

		while stage < #STAGES and x >= STAGES[stage + 1][1] do
			if stage > 0 then
				logLines[stage].done = true
			end
			stage = stage + 1
			status.Text = STAGES[stage][2] .. "..."
			logLines[stage] = { name = STAGES[stage][2], done = false }
			renderLog()
		end

		local pc = math.floor(p * 100 + 0.5)
		if pc ~= lastPct then
			lastPct = pc
			pct.Text = pc .. "%"
		end
		local sec = math.max(0, math.ceil(total - elapsed))
		if sec ~= lastSec then
			lastSec = sec
			hint.Text = skipText .. sec .. " s"
		end

		if now >= nextGlitch and elapsed > 2 then
			glitchUntil = now + 0.14
			glitchA, glitchB = math.random(1, nTitle), math.random(1, nTitle)
			nextGlitch = now + 2 + math.random() * 2.5
		end
		if now - lastTitle >= 0.05 then
			lastTitle = now
			buildTitle(now, elapsed)
		end

		if elapsed >= total then
			finish()
		end
	end)
end

	------------------------------------------------------------
	-- Lite mode: fewer glow layers & no breathing glow
	------------------------------------------------------------
	local function setLite(v, announce)
		if lite == v then
			return
		end
		lite = v
		for api in pairs(glows) do
			api:SetLimit(v and 1 or #api.list, 0.3)
		end
		if v then
			stopBreath()
		elseif isOpen then
			task.delay(0.4, function()
				if isOpen and not lite then
					startBreath()
				end
			end)
		end
		if announce then
			window:Notify(
				"Lite Mode",
				v and "Low FPS, glow effects reduced" or "FPS stable, glow effects restored",
				3,
				v and "warn" or "success"
			)
		end
	end

	function window:SetLite(v)
		autoLite = false
		setLite(v and true or false, false)
	end
	function window:IsLite()
		return lite
	end

		return fpsValue
	end

	local tabs, current = {}, nil

	-- search helpers
	local function countMatches(t, q)
		local n = 0
		for _, e in ipairs(t.entries) do
			if e.name ~= "" and string.find(e.name, q, 1, true) then
				n = n + 1
			end
		end
		return n
	end

	local function filterTab(t, q)
		if q == "" then
			for _, e in ipairs(t.entries) do
				e.frame.Visible = true
			end
			for _, g in ipairs(t.groups) do
				g.frame.Visible = true
				g.body.Visible = g.open
			end
			return
		end
		for _, g in ipairs(t.groups) do
			g.frame.Visible = false
			g.body.Visible = true
		end
		for _, e in ipairs(t.entries) do
			local hit = e.name ~= "" and string.find(e.name, q, 1, true) ~= nil
			e.frame.Visible = hit
			if hit and e.group then
				e.group.frame.Visible = true
			end
		end
	end

	local function styleTab(t, on)
		tw(t.button, 0.2, { BackgroundTransparency = on and 0.3 or 1 })
		tw(t.label, 0.2, { TextColor3 = on and T.text or T.sub })
		tw(t.bar, 0.2, { BackgroundTransparency = on and 0 or 1 })
		t.label.Font = on and Enum.Font.GothamBold or Enum.Font.GothamBold
	end

	local function selectTab(tab)
		if current == tab then
			return
		end
		local old = current
		current = tab
		for _, t in ipairs(tabs) do
			styleTab(t, t == tab)
		end
		filterTab(tab, query)

		local dir = 0
		if old then
			dir = (tab.index > old.index) and 1 or -1
			local o = old.page
			tw(o, 0.15, { GroupTransparency = 1 })
			task.delay(0.16, function()
				if current ~= old then
					o.Visible = false
				end
			end)
		end

		local p = tab.page
		p.GroupTransparency = 1
		p.Position = UDim2.fromOffset(0, dir * 14)
		p.Visible = true
		tw(p, 0.32, { GroupTransparency = 0, Position = UDim2.fromOffset(0, 0) })
	end

	searchBox:GetPropertyChangedSignal("Text"):Connect(function()
		query = string.lower(searchBox.Text)
		if query == "" then
			for _, t in ipairs(tabs) do
				filterTab(t, "")
			end
			return
		end
		if current and countMatches(current, query) == 0 then
			for _, t in ipairs(tabs) do
				if countMatches(t, query) > 0 then
					selectTab(t)
					break
				end
			end
		end
		if current then
			filterTab(current, query)
		end
	end)

	------------------------------------------------------------
	-- Ready: show the FPS bubble, open the window, start the FPS meter
	------------------------------------------------------------
	local function startFpsMeter()
		local frames, last = 0, os.clock()
		local lowN, highN, shown, tier = 0, 0, -1, 0

		-- per frame only: add 1 to a counter + read the clock once. Text updates every 0.5 s.
		connect(RunService.Heartbeat, function()
			frames = frames + 1
			local now = os.clock()
			local dt = now - last
			if dt < 0.5 then
				return
			end
			local fps = math.floor(frames / dt + 0.5)
			frames, last = 0, now
			fpsValue = fps

			if fps ~= shown then
				shown = fps
				fpsLbl.Text = fps .. ' <font color="' .. subHex .. '">FPS</font>'
			end

			local t = (fps >= 50) and 1 or ((fps >= 30) and 2 or 3)
			if t ~= tier then
				tier = t
				tw(fpsDot, 0.3, { BackgroundColor3 = TIER_COLORS[t] })
			end

			-- auto lite mode (with a delay so it doesn't flip back and forth)
			if autoLite then
				if fps < 25 then
					lowN, highN = lowN + 1, 0
				elseif fps >= 40 then
					highN, lowN = highN + 1, 0
				else
					lowN, highN = 0, 0
				end
				if not lite and lowN >= 4 then
					setLite(true, true)
				elseif lite and highN >= 6 then
					setLite(false, true)
				end
			end
		end)
	end

	-- called when the intro shutter starts opening: window & bubble appear
	local function opened()
		bubble.Visible = true
		setOpen(true)
		startFpsMeter()
	end

	-- called once the intro is completely gone: run the OnReady queue

		isReady = true
		for _, fn in ipairs(readyQ) do
			task.spawn(fn)
		end
	end

	if cfg.Intro == false then
		opened()
		introGone()
	else
		playIntro({
			title = title,
			subtitle = cfg.Subtitle,
			accent = accent,
			glow = glowColor,
			ink = onAccent,
			total = cfg.IntroTime or 20,
			skippable = cfg.Skippable,
			onOpen = opened,
			onDone = introGone,
		})
	end

	return window

-- the map is found through its CoinContainer (cached, searched at most once per second)
local boxCache, boxNext = nil, 0
local function findContainer()
	if boxCache and boxCache.Parent then
		return boxCache
	end
	boxCache = nil
	local now = os.clock()
	if now < boxNext then
		return nil
	end
	boxNext = now + 1
	boxCache = Workspace:FindFirstChild("CoinContainer", true)
	return boxCache
end

-- the dropped gun can be anywhere in the workspace (cached for half a second)
local gunCache, gunNext = nil, 0
local function findGunDrop()
	if gunCache and gunCache.Parent then
		return gunCache
	end
	gunCache = nil
	local now = os.clock()
	if now < gunNext then
		return nil
	end
	gunNext = now + 0.5
	local d = Workspace:FindFirstChild("GunDrop") or Workspace:FindFirstChild("GunDrop", true)
	if not d then
		-- a loose Gun tool lying in the world
		for _, o in ipairs(Workspace:GetChildren()) do
			if o:IsA("Tool") and o.Name == "Gun" then
				d = o
				break
			end
		end
	end
	gunCache = d
	return d
end


------------------------------------------------------------------
-- Role detection
-- 1) Knife / Gun tool in Backpack or Character
-- 2) fallback: the game's own GetPlayerData remote (if the tools are hidden)
------------------------------------------------------------------
local remoteData, remoteAt, remoteBusy = nil, 0, false
local function refreshRemote()
	if remoteBusy or os.clock() - remoteAt < 1.5 then
		return
	end
	remoteBusy = true
	task.spawn(function()
		pcall(function()
			local rf = ReplicatedStorage:FindFirstChild("GetPlayerData", true)
			if rf and rf:IsA("RemoteFunction") then
				local d = rf:InvokeServer()
				if type(d) == "table" then
					remoteData = d
				end
			end
		end)
		remoteAt = os.clock()
		remoteBusy = false
	end)
end

local function dataRole(plr)
	if not remoteData or os.clock() - remoteAt > 6 then
		return nil
	end
	local d = remoteData[plr.Name]
	if type(d) ~= "table" then
		return nil
	end
	local r = d.Role
	if r == "Murderer" then
		return "Murderer"
	elseif r == "Sheriff" or r == "Hero" then
		return "Sheriff"
	elseif r == "Innocent" then
		return "Innocent"
	end
	return nil
end

-- returns roles[player] = "Murderer" | "Sheriff" | "Innocent" (alive players only), murdererFound, container
local function scan()
	local roles, murd = {}, false
	for _, p in ipairs(Players:GetPlayers()) do
		if isAlive(p) then
			local r = false
			if hasTool(p, "Knife") then
				r = "Murderer"
				murd = true
			elseif hasTool(p, "Gun") then
				r = "Sheriff"
			end
			roles[p] = r
		end
	end
	-- other players' Backpacks are not visible to us, so while a round is running always
	-- fill the gaps from the game's own role data (not only when no murderer was found)
	local box = findContainer()
	if box then
		refreshRemote()
		for p, r in pairs(roles) do
			if not r then
				local dr = dataRole(p)
				if dr then
					roles[p] = dr
					if dr == "Murderer" then
						murd = true
					end
				end
			end
		end
	end
	for p, r in pairs(roles) do
		if not r then
			roles[p] = "Innocent"
		end
	end
	return roles, murd, box
end

local function namesOf(roles, role)
	local out = {}
	for p, r in pairs(roles) do
		if r == role then
			out[#out + 1] = p.DisplayName
		end
	end
	return table.concat(out, ", ")
end

local function announceRoles(roles)
	local m, s = namesOf(roles, "Murderer"), namesOf(roles, "Sheriff")
	if m == "" and s == "" then
		notify("Roles", "No roles detected yet", "warn")
		return
	end
	if m ~= "" then
		notify("Murderer", m, "error")
	end
	if s ~= "" then
		notify("Sheriff", s, "info")
	end
end

local myRole, announced, murdSince = nil, false, nil
local function tick()
	local roles, murd, box = scan()
	local active = murd or box ~= nil
	-- share the latest scan with the extra features (alert, HUD, teleports)
	F.roles, F.murd, F.box, F.rolesAt = roles, murd, box, os.clock()

	-- role notifications (only once weapons are handed out). Wait until the game's role data
	-- has arrived (or 3 s have passed) so the first announcement is complete.
	if murd then
		murdSince = murdSince or os.clock()
		local fresh = remoteData ~= nil and os.clock() - remoteAt < 6
		if fresh or os.clock() - murdSince > 3 then
			local mine = roles[LP]
			if mine and mine ~= myRole then
				myRole = mine
				if S.notifyRoles then
					if mine == "Murderer" then
						notify("Your role", "You are the Murderer", "error")
					elseif mine == "Sheriff" then
						notify("Your role", "You are the Sheriff", "info")
					else
						notify("Your role", "You are Innocent", "success")
					end
				end
			end
			if not announced then
				announced = true
				if S.notifyRoles then
					announceRoles(roles)
				end
			end
		end
	else
		myRole, announced, murdSince = nil, false, nil
	end

	-- player ESP
	local seen = {}
	if S.esp and active then
		for p, r in pairs(roles) do
			if p ~= LP and p.Character then
				local head = p.Character:FindFirstChild("Head")
				if head then
					local e = esp[p] or makeEsp(p)
					local col = COLORS[r]
					e.hl.Adornee = p.Character
					e.hl.FillColor = col
					e.hl.OutlineColor = col
					e.bb.Adornee = head
					e.tx.TextColor3 = col
					e.tx.Text = S.espNames and (p.DisplayName .. "  [" .. r .. "]") or r
					seen[p] = true
				end
			end
		end
	end
	for p in pairs(esp) do
		if not seen[p] then
			clearEsp(p)
		end
	end

	-- dropped gun ESP
	local gun = findGunDrop()
	if S.espGun and gun then
		if not gunHl then
			gunHl = Instance.new("Highlight")
			gunHl.FillColor = COLORS.Gun
			gunHl.OutlineColor = COLORS.Gun
			gunHl.FillTransparency = 0.4
			gunHl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
			gunHl.Parent = espFolder
		end
		gunHl.Adornee = gun
		gunHl.Enabled = true
	elseif gunHl then
		gunHl.Enabled = false
		gunHl.Adornee = nil
	end
end

task.spawn(function()
	while alive do
		pcall(tick)
		task.wait(0.25)
	end
end)

-- touch the gun with several body parts (executor touch + real contact)
local function touchGun(part)
	if not canTouch then
		return
	end
	local ch = LP.Character
	if not ch then
		return
	end
	local n = 0
	for _, p in ipairs(ch:GetChildren()) do
		if p:IsA("BasePart") then
			pcall(function()
				firetouchinterest(p, part, 0)
				firetouchinterest(p, part, 1)
			end)
			n = n + 1
			if n >= 5 then
				break
			end
		end
	end
end

local function grabGun()
	local gun = findGunDrop()
	if not gun then
		return false, "No dropped gun right now"
	end
	local root = getRoot()
	local part = partOf(gun)
	if not root or not part or not isAlive(LP) then
		return false, "You are not alive"
	end
	if hasTool(LP, "Knife") then
		return false, "The murderer can't pick up the gun"
	end
	if hasTool(LP, "Gun") then
		return true, "You already have the gun"
	end

	local back = root.CFrame
	local offsets = { 1, 2.5, 0, 3.5 }
	local got = false
	local t0, i = os.clock(), 0
	while os.clock() - t0 < 2.2 and alive do
		root = getRoot()
		if not root or not isAlive(LP) then
			break
		end
		i = i + 1
		-- stand on / inside the gun and touch it, a few heights in turn
		root.CFrame = CFrame.new(part.Position + Vector3.new(0, offsets[(i - 1) % #offsets + 1], 0))
		root.AssemblyLinearVelocity = Vector3.zero
		touchGun(part)
		task.wait(0.12)
		if hasTool(LP, "Gun") then
			got = true
			break
		end
		if not gun.Parent then
			-- the drop vanished: either we got it or someone else did
			task.wait(0.3)
			got = hasTool(LP, "Gun")
			break
		end
		part = partOf(gun) or part
	end
	root = getRoot()
	if root then
		root.CFrame = back
		root.AssemblyLinearVelocity = Vector3.zero
	end
	if got then
		return true, "Gun picked up"
	end
	return false, "Could not pick up the gun (someone may have taken it)"
end


task.spawn(function()
	local nextTry = 0
	while alive do
		if S.autoGun and os.clock() >= nextTry then
			local gun = findGunDrop()
			if gun and isAlive(LP) and not hasTool(LP, "Knife") and not hasTool(LP, "Gun") then
				nextTry = os.clock() + 3
				pcall(grabGun)
			end
		end
		task.wait(0.4)
	end
end)

	local function flyStep()
		local hum, root, cam = getHum(), getRoot(), Workspace.CurrentCamera
		if not (hum and root and cam) or hum.Health <= 0 then
			return
		end
		if not flyBV or flyBV.Parent ~= root then
			if flyBV then
				flyBV:Destroy()
			end
			if flyBG then
				flyBG:Destroy()
			end
			flyBV = Instance.new("BodyVelocity")
			flyBV.Name = "MatrixFlyV"
			flyBV.MaxForce = Vector3.new(1e9, 1e9, 1e9)
			flyBV.Velocity = Vector3.zero
			flyBV.Parent = root
			flyBG = Instance.new("BodyGyro")
			flyBG.Name = "MatrixFlyG"
			flyBG.MaxTorque = Vector3.new(1e9, 1e9, 1e9)
			flyBG.P = 1e4
			flyBG.D = 500
			flyBG.CFrame = root.CFrame
			flyBG.Parent = root
			hum.AutoRotate = false
		end
		local cf = cam.CFrame
		local dir = Vector3.zero
		if controls then
			local ok, mv = pcall(controls.GetMoveVector, controls)
			if ok and typeof(mv) == "Vector3" then
				dir = cf.RightVector * mv.X + cf.LookVector * -mv.Z
			end
		else
			dir = hum.MoveDirection
		end
		if dir.Magnitude < 0.05 then
			flyBV.Velocity = Vector3.zero
		else
			flyBV.Velocity = dir * FLY.speed
		end
		local look = Vector3.new(cf.LookVector.X, 0, cf.LookVector.Z)
		if look.Magnitude > 0.01 then
			flyBG.CFrame = CFrame.lookAt(root.Position, root.Position + look)
		end
	end

	local function httpGet(url)
		local ok, res = pcall(function()
			return game:HttpGet(url)
		end)
		if ok and type(res) == "string" and #res > 2 then
			return res
		end
		local fn = (typeof(request) == "function" and request)
			or (typeof(http_request) == "function" and http_request)
			or (typeof(syn) == "table" and syn.request)
			or nil
		if fn then
			local ok2, r = pcall(fn, { Url = url, Method = "GET" })
			if ok2 and type(r) == "table" and type(r.Body) == "string" then
				return r.Body
			end
		end
		return nil
	end

	local function fetchServers(smallest)
		local list, cursor = {}, nil
		for _ = 1, 3 do
			local url = string.format(
				"https://games.roblox.com/v1/games/%d/servers/Public?sortOrder=%s&excludeFullGames=true&limit=100",
				game.PlaceId,
				smallest and "Asc" or "Desc"
			)
			if cursor then
				url = url .. "&cursor=" .. cursor
			end
			local body = httpGet(url)
			if not body then
				break
			end
			local ok, data = pcall(function()
				return HttpService:JSONDecode(body)
			end)
			if not ok or type(data) ~= "table" or type(data.data) ~= "table" then
				break
			end
			for _, s in ipairs(data.data) do
				local playing, maxP = s.playing or 0, s.maxPlayers or 0
				if s.id ~= game.JobId and playing < maxP - 1 then
					list[#list + 1] = s
				end
			end
			cursor = data.nextPageCursor
			if not cursor or #list >= 30 then
				break
			end
		end
		return list
	end

	function F.hop(mode)
		if F.hopping then
			notify("Server Hop", "Already searching", "warn")
			return
		end
		F.hopping = true
		task.delay(15, function()
			F.hopping = false
		end)
		notify("Server Hop", "Searching for a server...", "info")
		task.spawn(function()
			local list = fetchServers(mode == "small")
			if #list == 0 then
				F.hopping = false
				notify("Server Hop", "No server found (request failed or blocked)", "error")
				return
			end
			local pick
			if mode == "small" then
				for _, s in ipairs(list) do
					if (s.playing or 0) >= 2 then
						pick = s
						break
					end
				end
				pick = pick or list[1]
			else
				local good = {}
				for _, s in ipairs(list) do
					if (s.playing or 0) >= 4 then
						good[#good + 1] = s
					end
				end
				local pool = (#good > 0) and good or list
				pick = pool[math.random(1, #pool)]
			end
			notify("Server Hop", "Joining a server with " .. (pick.playing or "?") .. " players", "success")
			local ok = pcall(function()
				TeleportService:TeleportToPlaceInstance(game.PlaceId, pick.id, LP)
			end)
			if not ok then
				F.hopping = false
				notify("Server Hop", "Teleport failed", "error")
			end
		end)
	end

	function F.rejoin()
		pcall(function()
			TeleportService:TeleportToPlaceInstance(game.PlaceId, game.JobId, LP)
		end)
	end

	local function findTool(name)

	local function equip(tool)
		local ch = LP.Character
		if tool and ch and tool.Parent ~= ch then
			local h = getHum()
			if h then
				pcall(function()
					h:EquipTool(tool)
				end)
			end
			local t0 = os.clock()
			while tool.Parent ~= ch and os.clock() - t0 < 0.5 do
				task.wait()
			end
		end
	end

	-- aim point on the target's torso, led by its movement
	-- speed = projectile speed (knife). nil = fixed prediction time (gun)
	local function aimAt(plr, speed)
		local ch = plr.Character
		local hrp = ch and ch:FindFirstChild("HumanoidRootPart")
		local part = ch and (ch:FindFirstChild("UpperTorso") or ch:FindFirstChild("Torso") or hrp)
		local root = getRoot()
		if not part or not hrp or not root then
			return nil
		end
		local v = hrp.AssemblyLinearVelocity
		v = Vector3.new(v.X, 0, v.Z)
		if v.Magnitude > 45 then
			v = v.Unit * 45
		end
		local dist = (part.Position - root.Position).Magnitude
		local t = speed and (dist / speed) or F.predict
		return part.Position + v * t, dist
	end

	------------------------------------------------------------
	-- Teleport to the nearest sheriff / innocent (+ auto mode)
	------------------------------------------------------------
	-- role = "Sheriff" | "Innocent" | "Any" (nearest of both)
	local function pickTarget(role)
		local roles = scan()
		if role == "Sheriff" or role == "Innocent" then
			return (nearestRole(roles, role))
		end
		local a, da = nearestRole(roles, "Innocent")
		local b, db = nearestRole(roles, "Sheriff")
		if a and b then
			return (da <= db) and a or b
		end
		return a or b
	end

	-- returns true, player when it teleported
	function F.tpNearest(role, silent)
		local p = pickTarget(role)
		if not p then
			if not silent then
				notify("Teleport", "No " .. ((role == "Any") and "target" or string.lower(role)) .. " found", "warn")
			end
			return false
		end
		local tr = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
		if not tr then
			return false
		end
		if tpTo(tr.CFrame * CFrame.new(0, 0, 3)) then
			if not silent then
				notify("Teleport", "Teleported to " .. p.DisplayName, "success")
			end
			return true, p
		end
		return false
	end

		AUTO.on = on and true or false

			AUTO.origin = root and root.CFrame or nil
			notify("Auto Teleport", "On. Use Teleport Back to return to where you started", "info")
		else
			AUTO.origin = nil
		end
	end

					if AUTO.origin then
						F.back = AUTO.origin -- "Teleport Back" returns to where auto started
					end

						notify("Auto Teleport", p.DisplayName, "info")
					end
				end)
			end
		end
	end)

	------------------------------------------------------------
	-- Aim Lock: gun = the camera follows the murderer,
	-- knife = the camera follows the nearest innocent or sheriff
	------------------------------------------------------------
	local lockTarget

	task.spawn(function()
		while alive do
			task.wait(0.1)
			lockTarget = nil
			if AL.on and isAlive(LP) then
				local p, d
				if hasTool(LP, "Gun") then
					-- sheriff: the murderer only
					p, d = nearestRole(F.roles, "Murderer")
				elseif hasTool(LP, "Knife") then
					-- murderer: the nearest innocent or sheriff
					local a, da = nearestRole(F.roles, "Innocent")
					local b, db = nearestRole(F.roles, "Sheriff")
					if a and b then
						if da <= db then
							p, d = a, da
						else
							p, d = b, db
						end
					else
						p, d = a or b, da or db
					end
				end
				if p and d <= AL.range then
					lockTarget = p
				end
			end
		end
	end)

	local function lockStep()
		if not AL.on or not lockTarget then
			return
		end
		local ch = lockTarget.Character
		local hrp = ch and ch:FindFirstChild("HumanoidRootPart")
		local part = ch and ((AL.part == "Head" and ch:FindFirstChild("Head")) or ch:FindFirstChild("UpperTorso") or ch:FindFirstChild("Torso") or hrp)
		local cam = Workspace.CurrentCamera
		if not part or not hrp or not cam then
			return
		end
		local v = hrp.AssemblyLinearVelocity
		v = Vector3.new(v.X, 0, v.Z)
		if v.Magnitude > 45 then
			v = v.Unit * 45
		end
		local aim = part.Position + v * F.predict
		cam.CFrame = cam.CFrame:Lerp(CFrame.lookAt(cam.CFrame.Position, aim), AL.strength)
	end

	pcall(function()
		RunService:BindToRenderStep("MatrixAimLock", Enum.RenderPriority.Camera.Value + 1, lockStep)
	end)

	function F.setLock(v, fromUI)
		AL.on = v and true or false
		F.quick.lock:SetActive(AL.on)
		F.quick.lock:SetSub(AL.on and "On" or "Off")
		if not fromUI and F.lockToggle then
			F.lockToggle:Set(AL.on, true)
		end
		if AL.on then
			notify("Aim Lock", "On. Gun locks the murderer, knife locks the nearest innocent or sheriff", "info")
		end
	end


	------------------------------------------------------------
	-- Fling (sit inside a player and spin hard) + Anti Fling
	------------------------------------------------------------
	local FLING = { busy = false, stop = false }
	local ANTI = { on = false, safe = nil, last = 0 }
	F.ANTI = ANTI

	-- mode: "Nearest" | "Murderer" | "Sheriff" | "Selected"
	local function flingTargetFor(mode)
		local roles = scan()
		if mode == "Murderer" then
			return (nearestRole(roles, "Murderer"))
		elseif mode == "Sheriff" then
			return (nearestRole(roles, "Sheriff"))
		elseif mode == "Selected" then
			return F.flingSel and F.findPlayer(F.flingSel) or nil
		end
		local root = getRoot()
		local best, bd
		for p in pairs(roles) do
			if p ~= LP and root then
				local tr = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
				if tr then
					local d = (tr.Position - root.Position).Magnitude
					if not bd or d < bd then
						best, bd = p, d
					end
				end
			end
		end
		return best
	end

	-- tap again while it runs to stop
	function F.fling(mode)
		if FLING.busy then
			FLING.stop = true
			notify("Fling", "Stopping...", "warn")
			return
		end
		local target = flingTargetFor(mode or F.flingMode)
		if not target then
			notify("Fling", "No target found", "warn")
			return
		end
		local root, hum = getRoot(), getHum()
		local tch = target.Character
		local tr = tch and tch:FindFirstChild("HumanoidRootPart")
		local th = tch and tch:FindFirstChildOfClass("Humanoid")
		if not root or not hum or hum.Health <= 0 or not tr or not th or th.Health <= 0 then
			notify("Fling", "Target not ready", "warn")
			return
		end

		FLING.busy, FLING.stop = true, false
		local back = root.CFrame
		local startPos = tr.Position
		local power = 10 ^ F.flingPower
		local flung = false
		notify("Fling", target.DisplayName, "info")
		hum.PlatformStand = true

		pcall(function()
			local t0 = os.clock()
			while alive and not FLING.stop and os.clock() - t0 < F.flingTime do
				root = getRoot()
				if not root or not tr.Parent or th.Health <= 0 then
					flung = th.Health <= 0
					break
				end
				if (tr.Position - startPos).Magnitude > 400 or tr.AssemblyLinearVelocity.Magnitude > 500 then
					flung = true
					break
				end
				-- stay inside the target (a little ahead of where it is going) and spin
				local v = tr.AssemblyLinearVelocity
				root.CFrame = tr.CFrame + Vector3.new(v.X, 0, v.Z) * 0.08
				root.AssemblyLinearVelocity = Vector3.new(power, power * 2, power)
				root.AssemblyAngularVelocity = Vector3.new(power, power, power)
				RunService.Heartbeat:Wait()
			end
		end)

		-- calm down and go back to where you were
		for _ = 1, 8 do
			root = getRoot()
			if root then
				root.AssemblyLinearVelocity = Vector3.zero
				root.AssemblyAngularVelocity = Vector3.zero
				root.CFrame = back
			end
			RunService.Heartbeat:Wait()
		end
		local h2 = getHum()
		if h2 then
			h2.PlatformStand = false
		end
		FLING.busy = false
		if flung then
			notify("Flung", target.DisplayName, "success")
		else
			notify("Fling", "No effect. Raise Fling Power or Fling Time", "warn")
		end
	end

	function F.setAntiFling(on)
		ANTI.on = on and true or false
		ANTI.safe = nil
	end

	-- other players' bodies stop colliding with you (nothing to push you with)
	track(RunService.Stepped:Connect(function()
		if not ANTI.on then
			return
		end
		for _, p in ipairs(Players:GetPlayers()) do
			if p ~= LP and p.Character then
				for _, part in ipairs(p.Character:GetChildren()) do
					if part:IsA("BasePart") and part.CanCollide then
						part.CanCollide = false
					end
				end
			end
		end
	end))

	-- if you still get launched: cancel the speed and go back to your last calm spot
	track(RunService.Heartbeat:Connect(function()
		if not ANTI.on or FLING.busy or FLY.on then
			return
		end
		local root = getRoot()
		if not root then
			return
		end
		local lv, av = root.AssemblyLinearVelocity, root.AssemblyAngularVelocity
		if lv.Magnitude > 250 or av.Magnitude > 60 then
			root.AssemblyLinearVelocity = Vector3.zero
			root.AssemblyAngularVelocity = Vector3.zero
			if ANTI.safe then
				root.CFrame = ANTI.safe
			end
		elseif os.clock() - ANTI.last > 0.2 and lv.Magnitude < 120 then
			ANTI.safe = root.CFrame
			ANTI.last = os.clock()
		end
	end))

	-- Links
	------------------------------------------------------------
	local function clipboardFn()
		return (typeof(setclipboard) == "function" and setclipboard)
			or (typeof(toclipboard) == "function" and toclipboard)
			or nil
	end

	function F.copyLink(url)
		url = url or F.tiktok
		local clip = clipboardFn()
		if clip and pcall(clip, url) then
			notify("Link copied", "Paste it in your browser", "success")
		else
			notify("Copy failed", url, "warn")
		end
	end

	function F.openLink(url)
		url = url or F.tiktok
		pcall(function()
			GuiService:OpenBrowserWindow(url)
		end)
		local clip = clipboardFn()
		if clip and pcall(clip, url) then
			notify("TikTok", "Link copied. Paste it in your browser if it did not open", "success")
		else
			notify("TikTok", url, "info")
		end
	end

	------------------------------------------------------------
	-- Cleanup (called when the UI is closed)
	------------------------------------------------------------
	function F.cleanup()
		FLY.on = false
		AL.on = false
		AUTO.on = false
		ANTI.on = false
		FLING.stop = true
		pcall(function()
			RunService:UnbindFromRenderStep("MatrixAimLock")
		end)
		killFly()
		pcall(function()
			hud:Destroy()
		end)
		if safePart then
			pcall(function()
				safePart:Destroy()
			end)
		end
	end
end

------------------------------------------------------------------
-- Coin farm loop
------------------------------------------------------------------
local lastStatus = ""
local function setStatus(t)
	if t ~= lastStatus then
		lastStatus = t
		farmStatus:Set("Status: " .. t)
	end
end

local function nextCoin(box, visited, from)
	local best, bestD
	for _, c in ipairs(box:GetChildren()) do
		if c:IsA("BasePart") and not visited[c] and string.find(c.Name, "Coin") then
			local d = (c.Position - from).Magnitude
			if not bestD or d < bestD then
				best, bestD = c, d
			end
		end
	end
	return best
end

-- smooth straight-line move; stops if farming is turned off or the character changes
local function moveTo(pos, speed)
	local ch = LP.Character
	local root = getRoot()
	if not root then
		return false
	end
	local start = root.Position
	local dur = math.max((pos - start).Magnitude / speed, 0.05)
	local t = 0
	while t < dur do
		t = t + RunService.Heartbeat:Wait()
		root = getRoot()
		if not alive or not S.farm or not root or LP.Character ~= ch then
			return false
		end
		local a = math.min(t / dur, 1)
		root.CFrame = CFrame.new(start:Lerp(pos, a)) * root.CFrame.Rotation
		root.AssemblyLinearVelocity = Vector3.zero
	end
	return true
end

task.spawn(function()
	local lastBox, visited, count, resetDone = nil, {}, 0, false
	while alive do
		if not S.farm then
			farming = false
			setStatus("off")
			task.wait(0.4)
		else
			local box = findContainer()
			local hum, root = getHum(), getRoot()
			if not box then
				setStatus("waiting for a round")
				task.wait(0.6)
			elseif not (hum and root and hum.Health > 0) then
				setStatus("waiting for respawn")
				task.wait(0.6)
			else
				if box ~= lastBox then
					lastBox, visited, count, resetDone = box, {}, 0, false
				end
				if count >= S.farmLimit then
					farming = false
					setStatus("limit reached (" .. count .. ")")
					if S.farmReset and not resetDone then
						resetDone = true
						hum.Health = 0
					end
					task.wait(0.8)
				else
					local coin = nextCoin(box, visited, root.Position)
					if not coin then
						farming = false
						setStatus("no coins left")
						task.wait(0.5)
					else
						setStatus("collecting " .. (count + 1) .. "/" .. S.farmLimit)
						farming = true
						local ok = moveTo(coin.Position, S.farmSpeed)
						if ok then
							touchPart(coin)
							task.wait(0.12)
							visited[coin] = true
							count = count + 1
							F.stats.coins = F.stats.coins + 1
						end
						task.wait()
					end
				end
			end
		end
	end
end)

-- Movement / misc loops
------------------------------------------------------------------
track(RunService.Heartbeat:Connect(function()
	local hum = getHum()
	if not hum then
		return
	end
	if S.speedOn and hum.WalkSpeed ~= S.speed then
		hum.WalkSpeed = S.speed
	end
	if S.jumpOn then
		if not hum.UseJumpPower then
			hum.UseJumpPower = true
		end
		if hum.JumpPower ~= S.jump then
			hum.JumpPower = S.jump
		end
	end
end))

track(RunService.Stepped:Connect(function()
	if not (S.noclip or farming) then
		return
	end
	local ch = LP.Character
	if not ch then
		return
	end
	for _, part in ipairs(ch:GetChildren()) do
		if part:IsA("BasePart") and part.CanCollide then
			part.CanCollide = false
		end
	end
end))

track(UserInputService.JumpRequest:Connect(function()
	if S.infJump then
		local hum = getHum()
		if hum then
			hum:ChangeState(Enum.HumanoidStateType.Jumping)
		end
	end
end))

track(LP.Idled:Connect(function()
	if S.antiAfk then
		pcall(function()
			VirtualUser:CaptureController()
			VirtualUser:ClickButton2(Vector2.new())
		end)
	end
end))

-- Cleanup when the UI is closed
------------------------------------------------------------------
local origDestroy = win.Destroy
function win:Destroy()
	alive = false
	farming = false
	S.farm = false
	pcall(F.cleanup)
	for _, c in ipairs(conns) do
		pcall(function()
			c:Disconnect()
		end)
	end
	for p in pairs(esp) do
		clearEsp(p)
	end
	pcall(function()
		espFolder:Destroy()
	end)
	local h = getHum()
	if h then
		if S.speedOn then
			h.WalkSpeed = 16
		end
		if S.jumpOn then
			h.JumpPower = 50
		end
	end
	if env.MatrixMM2 == win then
		env.MatrixMM2 = nil
	end
	origDestroy(self)
end
